#include <cstdio>
class Entite{
public:
    virtual const char* nom() const{
        return "entite";
    }
    virtual ~Entite(){}
};
class Ennemi:public Entite{
public:
    int degats;
    const char* nom() const override{
        return "ennemi";
    }
};
void f(Entite e){
    printf("%s\n",e.nom());
}
void g(Entite& e){
    printf("%s\n",e.nom());
}
void h(Entite* e){
    printf("%s\n",e->nom());
}
int main(){
    Ennemi e;
    f(e);
    g(e);
    h(&e);
    printf("%zu\n",sizeof(Entite));
    printf("%zu\n",sizeof(Ennemi));

    return 0;
}
