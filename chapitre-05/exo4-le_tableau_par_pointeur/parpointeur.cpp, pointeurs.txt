
#include <cstdio>
int somme(const int t[], int n){
    int resultat = 0;
    const int* p=t;
     for (int i=0; i<n;i++){
        resultat += *p;
        p++;
    }
     return resultat;
}
void minMax(const int t[], int n, int& min, int& max){
    const int* p=t;
    min = *p;
    max = *p;
    p++;
    for (int i=1;i<n;i++){
        if (*p<min){
            min=*p;
        }
        if (*p>max){
            max=*p;
        }
        p++;
    }
}
void renverser(int t[], int n){
    int* gauche=t;
    int* droite=t+n-1;
    while (gauche<droite) {
        int temporaire= *gauche;
        *gauche= *droite;
        *droite=temporaire;
        gauche++;
        droite--;
    }
}
int longueur(const char* chaine){
    const char* p = chaine;
     while (*p != '\0'){
        p++;
     }
      return static_cast<int>(p - chaine);
}
