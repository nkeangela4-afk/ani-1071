 
#include <cstdio>
#include <cmath>
using Fonction = double (*)(double);
double integrale(Fonction f, double a, double b) {
    double pas = (b - a) / 1000.0;
    double aire = 0.0;
    for (int i = 0; i < 1000; i++) {
        aire += f(a + i * pas) * pas;
    }
    return aire;
}
double derivee(Fonction f, double x) {
    double h = 1e-6;
    return (f(x + h) - f(x - h)) / (2 * h);
}
double zero(Fonction f, double a, double b) {
    double fa = f(a);
    double fb = f(b);
    if (fa * fb > 0) {
        return NAN;
    }
    while (b - a > 1e-6) {
        double milieu = (a + b) / 2;
        double fm = f(milieu);
        if (fm == 0) {
            return milieu;
        }
        if (fa * fm < 0) {
            b = milieu;
            fb = fm;
        } else {
            a = milieu;
            fa = fm;
        }
    }
    return (a + b) / 2;
}
double sinus(double x) {
    return sin(x);
}
double carre(double x) {
    return x * x;
}
double fonction_zero(double x) {
    return x * x - 2;
