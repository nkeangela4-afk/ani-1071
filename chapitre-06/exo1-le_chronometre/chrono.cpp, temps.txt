#include <chrono>
#include <cstdio>
using namespace std;
using namespace chrono;
double maintenant(){
    auto t = steady_clock::now().time_since_epoch();
    return duration<double>(t).count();
}
int main(){
    long long total = 0;
    double debut = maintenant();
    for (int i = 1; i <= 1000000; ++i){
        total += i;
    }
    double fin = maintenant();
    double duree_boucle = fin - debut;
    printf("Total = %lld\n", total);
    printf("Duree de la boucle : %.9f secondes\n", duree_boucle);
    double durees[5];
    for (int i = 0; i < 5; ++i){
        debut = maintenant();
        printf("Bonjour !\n");
        fin = maintenant();
        durees[i] = fin - debut;
    }
    FILE *f = fopen("temps.txt", "w");
    if (f == nullptr){
        printf("Erreur lors de la creation de temps.txt\n");
        return 1;
    }
    fprintf(f, "-O0 :\n");
    fprintf(f, "Total = %lld\n", total);
    fprintf(f, "Duree de la boucle = %.9f secondes\n", duree_boucle);
    fprintf(f, "\n-O2 :\n");
    fprintf(f, "Total = %lld\n", total);
    fprintf(f, "Duree de la boucle = %.9f secondes\n", duree_boucle);
    fprintf(f, "\nUn printf :\n");
    for (int i = 0; i < 5; ++i){
        fprintf(f, "Duree %d = %.9f secondes\n", i + 1, durees[i]);
    }
    fprintf(f, "\nLe plus cher :\n");
    fprintf(f, "Comparer les durees mesurees.\n");
    fclose(f);

    return 0;
}
