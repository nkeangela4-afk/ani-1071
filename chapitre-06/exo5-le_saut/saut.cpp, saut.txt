
#include <cstdio>
#include <cmath>
struct Vec2{
    double x;
    double y;
};
void eulerSemiImplicite(Vec2& p, Vec2& v, Vec2 a, double dt){
    v.x = v.x + a.x * dt;
    v.y = v.y + a.y * dt;
    p.x = p.x + v.x * dt;
    p.y = p.y + v.y * dt;
}
double simuler(double dt){
    double g = 9.81;
    double hMax = 0;
    double v0 = (2 * g * 3);
    Vec2 p = {0, 0};
    Vec2 v = {0, v0};
    Vec2 a = {0, -g};
    while (p.y >= 0) {
        eulerSemiImplicite(p, v, a, dt);
        if (p.y > hMax){
            hMax = p.y;
        }
    }
    return hMax;
}
int main(){
    double g = 9.81;
    double h = 3;
    double v0 = sqrt(2 * g * h);
    double h120 = simuler(1.0 / 120.0);
    double h30 = simuler(1.0 / 30.0);
    double dt = 1.0 / 1000.0;
    double hPetit = simuler(dt);
    FILE* fichier = fopen("saut.txt", "w");
    if (fichier == NULL) {
        printf("Impossible de creer saut.txt\n");
        return 1;
    }
    fprintf(fichier, "v0 : %.6f m/s\n", v0);
    fprintf(fichier, "1/120 : %.6f m\n", h120);
    fprintf(fichier, "1/30 : %.6f m\n", h30);
    fprintf(fichier, "Mon pas : dt = %.6f s, hauteur = %.6f m\n",dt, hPetit);
    fprintf(fichier,"L'ecart vient : de la simulation numerique et du pas de temps. ""La formule donne 3 m, mais Euler semi-implicite sous-estime ""la hauteur maximale. Un pas plus petit reduit cet ecart.\n");
    fclose(fichier);
    printf("Vitesse initiale : %.6f m/s\n", v0);
    printf("Hauteur avec dt = 1/120 : %.6f m\n", h120);
    printf("Hauteur avec dt = 1/30 : %.6f m\n", h30);
    printf("Hauteur avec dt = %.6f : %.6f m\n", dt, hPetit);
    printf("Resultats enregistres dans saut.txt\n");

    return 0;
}
